구글 시트에 바로 저장 (Google Apps Script 웹앱) 설정

① 구글 시트 → 확장 프로그램 → Apps Script 에 아래 코드를 붙여넣고 저장 → ② 배포 → 새 배포 → 유형 ‘웹 앱’ → 실행 계정 ‘나’, 액세스 ‘모든 사용자’ → 배포 → ③ 나온 웹앱 URL(/exec)을 아래에 붙여넣고 저장하세요. 

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  var ss = body.sheetUrl
    ? SpreadsheetApp.openByUrl(body.sheetUrl)
    : SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(body.sheetName || '강의일정') || ss.getActiveSheet();
  var rows = body.rows || [];
  if (rows.length) {
    sh.getRange(sh.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows);
  }
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, added: rows.length }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  return ContentService.createTextOutput('ok');
}
